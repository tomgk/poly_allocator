11/**
12 * @brief A growth-based arena allocator that stores objects in contiguous memory.
13 *
14 * ArenaAllocator manages objects in a std::vector<std::byte> with the following characteristics:
15 * - Objects are stored contiguously with proper alignment
16 * - When new allocation doesn't fit, the buffer doubles in size
17 * - All alive objects are moved to the new buffer with proper construction/destruction
18 * - Destroyed objects leave gaps but remain in place (marked as dead)
19 * - Each allocation tracks whether it's still alive via a flag
20 * - Optional runtime type information storage
21 * - Forward iteration support over alive allocations
22 *
23 * @tparam StoreTypeInfo If true, stores type information for each allocation. Default: false
24 *
25 * @note When StoreTypeInfo is false, no runtime overhead is incurred for type tracking.
26 */
27template <bool StoreTypeInfo = false>
28class ArenaAllocator {
29public:
30 /**
31 * @brief Forward iterator for traversing alive allocations in the arena.
32 *
33 * The iterator automatically skips dead allocations and provides access to
34 * allocation metadata. Iterators become invalid if the arena is reallocated
35 * during iteration.
36 */
37 class Iterator {
38 public:
39 using difference_type = std::ptrdiff_t;
40 using value_type = void*;
41 using pointer = void**;
42 using reference = void*&;
43 using iterator_category = std::forward_iterator_tag;
44

46 friend class ArenaAllocator;

51 /**
52 * @brief Construct iterator at given byte offset.
53 * @param arena Pointer to the arena
54 * @param offset Byte offset in the buffer
55 * @param find_alive If true, advance to first alive allocation
56 */
57 Iterator(const ArenaAllocator* arena, std::size_t offset, bool find_alive)
58 : arena(arena), offset(offset) {
59 if (find_alive) {
60 advance_to_next_alive();
61 }
62 }
63
64 /**
65 * @brief Advance offset to the next alive allocation.
66 */
67 void advance_to_next_alive() {
68 while (offset < arena->current_offset) {
69 const AllocationHeader& header = arena->get_header(offset);
70 if (header.is_alive) {
71 break;
72 }
73 offset += HEADER_SIZE + header.size;
74 }
75 }

106 * @return Reference to this iterator after advancing

111 offset += HEADER_SIZE + header.size;

115 }

145 /**
146 * @brief Metadata stored before each allocation.
147 */
148 struct AllocationHeader {
149 std::size_t size; ///< Size of the actual object (excluding header)
150 bool is_alive; ///< Flag indicating if object is still alive
151 void (*destructor)(void*); ///< Function pointer to object's destructor
152 std::conditional_t<StoreTypeInfo, const std::type_info*, struct {} > type_info; ///< Optional: RTTI information
153
154 AllocationHeader() : size(0), is_alive(false), destructor(nullptr) {
155 if constexpr (StoreTypeInfo) {
156 type_info = nullptr;
157 }
158 }
159 };
160
161 static constexpr std::size_t INITIAL_CAPACITY = 256; ///< Initial buffer capacity
162 static constexpr std::size_t HEADER_SIZE = sizeof(AllocationHeader);
163
164 std::vector<std::byte> buffer; ///< Raw byte buffer storing all allocations
165 std::size_t current_offset = 0; ///< Byte offset to end of last allocation

167 /**
168 * @brief Align an offset to the required alignment boundary.
169 * @param offset Current offset
170 * @param alignment Required alignment in bytes
171 * @return Aligned offset

173 static std::size_t align_offset(std::size_t offset, std::size_t alignment) {
174 return (offset + alignment - 1) & ~(alignment - 1);
175 }

213 /**
214 * @brief Reallocate to a larger buffer and move all alive objects.
215 *
216 * Doubles capacity until the new allocation fits. All alive objects are
217 * moved and destructors of old objects are called.
218 *
219 * @param required_size Minimum additional bytes needed
220 */
221 void reallocate(std::size_t required_size) {
222 std::size_t new_capacity = buffer.capacity();
223
224 // Double capacity until it fits
225 while (new_capacity < current_offset + required_size) {
226 new_capacity *= 2;
227 }
228
229 // Create new buffer
230 std::vector<std::byte> new_buffer;
231 new_buffer.reserve(new_capacity);
232 new_buffer.resize(0);
233
234 std::size_t new_offset = 0;
235
236 // Move all alive objects to the new buffer
237 std::size_t offset = 0;
238 while (offset < current_offset) {
239 AllocationHeader& old_header = get_header(offset);
240 std::size_t object_size = old_header.size;
241
242 if (old_header.is_alive) {
243 // Align the new offset
244 std::size_t aligned_new_offset = align_offset(new_offset, alignof(AllocationHeader));
245
246 // Ensure new_buffer has enough space
247 if (new_buffer.capacity() < aligned_new_offset + HEADER_SIZE + object_size) {
248 new_buffer.reserve(std::max(new_buffer.capacity() * 2,
249 aligned_new_offset + HEADER_SIZE + object_size + 256));
250 }
251 new_buffer.resize(aligned_new_offset + HEADER_SIZE + object_size);
252
253 // Copy header
254 AllocationHeader& new_header = *reinterpret_cast<AllocationHeader*>(
255 new_buffer.data() + aligned_new_offset);
256 new_header = old_header;
257
258 // Copy object data
259 std::memcpy(new_buffer.data() + aligned_new_offset + HEADER_SIZE,
260 get_object_pointer(offset),
261 object_size);
262
263 new_offset = aligned_new_offset + HEADER_SIZE + object_size;
264 }
265
266 // Move to next allocation
267 offset += HEADER_SIZE + object_size;
268 }
269
270 // Destroy all objects in the old buffer
271 offset = 0;
272 while (offset < current_offset) {
273 AllocationHeader& header = get_header(offset);
274 std::size_t object_size = header.size;
275
276 if (header.is_alive && header.destructor) {
277 header.destructor(get_object_pointer(offset));
278 }
279
280 offset += HEADER_SIZE + object_size;
281 }
282
283 // Replace old buffer with new one
284 buffer = std::move(new_buffer);
285 current_offset = new_offset;
286 }

296 /**
297 * @brief Allocate memory for an object of type T and construct it.
298 *
299 * Allocates space for an object of type T, properly aligned, with an
300 * allocation header. If necessary, reallocates the buffer.
301 *
302 * @tparam T Type of object to allocate
303 * @tparam Args Types of constructor arguments
304 * @param args Arguments to forward to T's constructor
305 * @return Pointer to the newly constructed object
306 *
307 * @note The returned pointer is valid until the next reallocation
308 */
309 template <typename T, typename... Args>
310 T* allocate(Args&&... args) {
311 // Calculate required space: header + alignment padding + object
312 std::size_t header_offset = current_offset;
313 std::size_t object_offset = align_offset(header_offset + HEADER_SIZE, alignof(T));
314 std::size_t required_size = object_offset - header_offset + sizeof(T);
315
316 // Check if reallocation is needed
317 if (current_offset + required_size > buffer.capacity()) {
318 reallocate(required_size);
319 // Recalculate offsets after reallocation
320 header_offset = current_offset;
321 object_offset = align_offset(header_offset + HEADER_SIZE, alignof(T));
322 }
323
324 // Ensure buffer is large enough
325 if (buffer.size() < object_offset + sizeof(T)) {
326 buffer.resize(object_offset + sizeof(T));
327 }
328
329 // Place header
330 AllocationHeader& header = *reinterpret_cast<AllocationHeader*>(
331 buffer.data() + header_offset);
332 header.size = sizeof(T);
333 header.is_alive = true;
334 header.destructor = [](void* ptr) {
335 reinterpret_cast<T*>(ptr)->~T();
336 };
337
338 // Store type information if enabled
339 if constexpr (StoreTypeInfo) {
340 header.type_info = &typeid(T);
341 }
342
343 // Construct object in place with forwarded arguments
344 T* obj = new (buffer.data() + object_offset) T(std::forward<Args>(args)...);
345
346 // Update offset
347 current_offset = object_offset + sizeof(T);
348
349 return obj;
350 }

352 /**
353 * @brief Deallocate an object previously allocated via allocate().
354 *
355 * Calls the object's destructor and marks the allocation as dead.
356 * The space remains in the buffer until the next reallocation.
357 *
358 * @tparam T Type of object to deallocate
359 * @param ptr Pointer to object returned by allocate()
360 *
361 * @note Must be called with a pointer returned by allocate() on this allocator
362 */
363 template <typename T>
364 void deallocate(T* ptr) {
365 if (!ptr) return;
366
367 // Find the object and mark it as dead, but don't move anything
368 std::size_t offset = 0;
369 while (offset < current_offset) {
370 AllocationHeader& header = get_header(offset);
371 std::size_t object_size = header.size;
372 std::byte* object_ptr = get_object_pointer(offset);
373
374 if (object_ptr == reinterpret_cast<std::byte*>(ptr)) {
375 // Call destructor and mark as dead
376 if (header.is_alive && header.destructor) {
377 header.destructor(object_ptr);
378 }
379 header.is_alive = false;
380 return;
381 }
382
383 offset += HEADER_SIZE + object_size;
384 }
385
386 assert(false && "Object not found in arena");
387 }

389 /**
390 * @brief Get runtime type information for an allocated object.
391 *
392 * @tparam T Type of object
393 * @param ptr Pointer to object returned by allocate()
394 * @return Pointer to std::type_info for the allocated object, or nullptr if not found
395 *
396 * @note This method is only available if StoreTypeInfo template parameter is true
397 */
398 template <typename T>
399 const std::type_info* get_type_info(T* ptr) const
400 requires StoreTypeInfo
401 {
402 if (!ptr) return nullptr;
403
404 std::size_t offset = 0;
405 while (offset < current_offset) {
406 AllocationHeader& header = get_header(offset);
407 std::size_t object_size = header.size;
408 std::byte* object_ptr = get_object_pointer(offset);
409
410 if (object_ptr == reinterpret_cast<std::byte*>(ptr)) {
411 return header.type_info;
412 }
413
414 offset += HEADER_SIZE + object_size;
415 }
416
417 return nullptr;
418 }
